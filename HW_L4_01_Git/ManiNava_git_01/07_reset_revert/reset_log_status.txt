## restore --staged: unstage
$ echo "Modified" > file.txt
$ git add file.txt
$ git restore --staged file.txt
$ git status --short
 M file.txt

## restore: discard changes in working tree
$ git restore file.txt
$ git status --short

## reset --soft HEAD~1 (moves HEAD only)

9c81855 Commit 1
$ git reset --soft HEAD~1
$ git log --oneline
65a10b5 Commit 2
9c81855 Commit 1
$ git status

nothing to commit, working tree clean
