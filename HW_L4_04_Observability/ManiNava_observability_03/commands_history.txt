=== 01_alertmanager_setup: stack started ===
# loaded rules
curl -s http://localhost:9090/api/v1/rules | jq -c '.data.groups[] | {name: .name, rules: [.rules[].name]}'
# active alertmanager connection
curl -s http://localhost:9090/api/v1/alertmanagers | jq -c '.data.activeAlertmanagers[].url'
# targets
curl -s http://localhost:9090/api/v1/targets | jq '.data.activeTargets[] | {job: .labels.job, health: .health}'
# promtool check config
docker exec prometheus promtool check config /etc/prometheus/prometheus.yml
# amtool check alertmanager config
docker exec alertmanager amtool check-config /etc/alertmanager/alertmanager.yml
=== 01_alertmanager_setup done ===
=== demo_instance_down start ===
# up query before
curl -s 'http://localhost:9090/api/v1/query?query=up%7Bjob%3D%22node_exporter%22%7D' | jq -c '.data.result[] | {metric, value}'
docker stop node_exporter
# prometheus alerts after firing
curl -s http://localhost:9090/api/v1/alerts | jq -c '.data.alerts[] | {alertname: .labels.alertname, state, value: .value, activeAt: .activeAt}'
# alertmanager received the alert
curl -s http://localhost:9093/api/v2/alerts | jq -c '.[] | {labels: {alertname: .labels.alertname, severity: .labels.severity, instance: .labels.instance}, status: .status.state}'
docker compose up -d node_exporter (heal)
# up query after restart
curl -s 'http://localhost:9090/api/v1/query?query=up%7Bjob%3D%22node_exporter%22%7D' | jq -c '.data.result[] | {metric, value}'
