docker pull nginx:alpine

# S2.1 basic nginx container
docker run -d --name nginx-basic -p 8080:80 nginx:alpine
docker logs nginx-basic | head -n 10 > nginx_logs.txt
docker inspect nginx-basic | grep -A 5 "NetworkSettings" > nginx_network.txt
docker stop nginx-basic
docker rm nginx-basic
# S2.2 static bind-mount site
docker run -d --name nginx-static \
docker stop nginx-static
docker rm nginx-static
# S2.3 nginx reverse proxy
docker network create rp-net > /dev/null 2>&1 || true
docker run -d --name backend-api \
docker run -d --name nginx-proxy \
docker logs nginx-proxy | tail -n 10 > proxy_logs.txt
docker stop nginx-proxy backend-api
docker rm nginx-proxy backend-api
docker network rm rp-net > /dev/null 2>&1 || true

# S3.2 https server + http_vs_https

docker logs nginx-https | tail -n 10 > https_logs.txt

docker stop nginx-https
docker rm nginx-https
# S4.1 stub_status endpoint
docker run -d --name nginx-status \

# S4.2 log monitoring
docker run -d --name nginx-logs \
docker logs nginx-logs --tail 20 > latest_logs.txt
docker logs nginx-logs 2>&1 | grep -i error > error_logs.txt || echo "No errors found" > error_logs.txt
docker logs nginx-logs --since 1m > recent_logs.txt
docker stop nginx-logs
docker rm nginx-logs

docker network create hb-net > /dev/null 2>&1 || true
docker run -d --name backend1 \

docker run -d --name haproxy-lb \
